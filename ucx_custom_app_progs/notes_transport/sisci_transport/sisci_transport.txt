UCS_CLASS_INIT_FUNC _for_uct_sci_iface_t (md, worker, iface_params, iface_config):
    - sci_md = derived_of(md)             // sci_md.super = md, reverse of that
    - config = derived_of(iface_config)   // iface_config.super = config, reverse of that

    // assert UCT_IFACE_PARAM_FIELD_OPEN_MODE bit set in params.field_mask

    // priv_worker = derived_of(worker)
    // assert priv_worker.thread_mode != UCS_THREAD_MODE_MULTI
    // Make sure sisci transport is used by a single threaded worker, it does not support multi-threaded workers

    // init lock that is used by connection handler
    - mutex_init(self.lock)

    - UCS_CLASS_CALL_SUPER_INIT_for_uct_base_iface_t(
        uct_sci_iface_ops,
        uct_base_iface_internal_ops,
        md,
        worker,
        params,
        tl_config
        UCS_STATS_ARG(...)
      )

    - SCIGetLocalNodeId(adapter_id, flags) < node_id >

    // self here seems to point to sci_iface... No idea why this is a thing
    self.device_addr = nodeID;
    self.segment_id  = ucs_generate_uuid(trash);
    self.ctl_id      = ucs_generate_uuid(trash);
    self.send_size   = config.send_size;
    self.eps         = 0;
    self.max_eps     = MIN(SCI_MAX_EPS, config.max_eps);
    self.connections = 0;
    self.queue_size  = config.queue_size;

    - SCIOpen < self.sd_for_endpoints >

    - SCIOpen < self.sd_for_control >

    // Create a "receive segment"
    recv_segment_size = self.send_size * self.max_eps * self.queue_size
    - SCICreateSegment (sci_md.sci_virtual_device, self.segment_id, recv_segment_size) < self.local_segment >
    - SCIPrepareSegment < self.local_segment >
    - SCISetSegmentAvailable < self.local_segment >
    - SCIMapLocalSegment(self.local_segment, recv_segment_size) < self.tx_buf >

    // Create a "control segment"
    control_segment_size = sizeof(sci_ctl_t) * self.max_eps
    - SCICreateSegment(sci_md.sci_virtual_device, self.ctl_id, control_segment_size) < self.ctl_segment >
    - SCIPrepareSegment < self.ctl_segment >
    - SCISetSegmentAvailable < self.ctl_segment >
    - SCIMapLocalSegment(self.ctl_segment, control_segment_size) < self.ctls >

    // DMA
    - SCICreateDMAQueue(sci_md.sci_virtual_device) < self.dma_queue >
    dma_seg_id = ucs_generate_uuid(trash);
    - SCICreateSegment(sci_md.sci_virtual_device, dma_seg_id, self.send_size) < self.dma_segment >
    - SCIPrepareSegment < self.dma_segment >
    - SCIMapLocalSegment(self.dma_segment, self.send_size) < self.dma_buf >

    self->interrupt_no = ucs_generate_uuid(trash);
    - SCICreateDataInterrupt(sci_md.sci_virtual_device, self.interrupt_no, callback=conn_handler, callback_arg=self) < self.interrupt >
    

// uninteresting functions start

// establishes connection with endpoint
conn_handler(arg, interrupt, data, length)

    - sci_iface = (uct_sci_iface_t*) arg;
    - connection_request = (conn_req_t*) data;
    - md = derived_of(sci_iface.super.md)

    - SCIConnectDataInterrupt(md.sci_virtual_device, connection_request.node_id, connection_request.interrupt_no) < ans_interrupt >
    
    - with lock sci_iface.lock

        - get sci_cd from sci_iface.sci_cds array where sci_iface->sci_cds[i].status == 0
          then set sci_iface->sci_cds[i].status = 2
          connection_descriptor = sci_iface->sci_cds[i]

        - sci_iface.connections++

    // answer is a struct with these variables
    answer.node_id = sci_iface.device_addr
    answer.segment_id = sci_iface.segment_id
    answer.offset = connection_descriptor
    answer.send_size = sci_iface.send_size
    answer.queue_size = sci_iface.queue_size

    - SCITriggerDataInterrupt(ans_interrupt, answer)

    - SCIConnectSegment(sci_iface.vdev_ctl, connection_request.node_id, connection_request.ctl_seg_id) < connection_descriptor.ctl_segment >

    - SCIMapRemoteSegment(connection_descriptor.ctl_segment, ctl_segment.map, ) < connection_descriptor.ctl_buffer >

    - SCIDisconnectDataInterrupt(ans_interrupt)





uct_sci_iface_progress(tl_iface)
    - sci_iface = derived_of(tl_iface)
    - while (more_ready_messages)
        - for all connections in sci_iface.connections
            
            if (anything out of the ordinary): continue, i.e. skip loop iteration

            // cd is short for connection descriptor or something
            - cd = sci_iface
            
            // send_size is obviously the length of a "packet" kind of
            // and queue_size refers to the number of packets that the send queue can hold, not the size in bytes
            - offset = sci_iface.send_size * ((cd.last_ack+1) % sci_iface.queue_size)
            - packet = cd.cd_buf + offset
            - data_ptr = packet + sizeof(sci_packet_t)

            // check whether packet has arrived
            if (packet.status != 1): continue, i.e. skip loop iteration

            - status = uct_iface_invoke_am(sci_iface.super, packet.am_id, data_ptr, packet.length)

            if (status is okay, not in progress)
                packet.status = 0;
                cd.ctl_buf.status = 0;
                cd.last_ack++
                cd.ctl_buf.ack = cd.last_ack; 
                SCIFlush
                more_ready_messages = false

    return count
