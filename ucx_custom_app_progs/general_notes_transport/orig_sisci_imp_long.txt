uct_sci_ep_t* ep
First gets a pointer to an endpoint struct  (remote segment, where to send to), Used to get some offset related things. Send size, sequence number, queue size. Whatever those things are.

sci_packet_t* packet;
A pointer to a packet struct (Only meta info apparently. status, am_id, length).

ucs_status_t uct_sci_ep_am_short(uct_ep_h tl_ep, uint8_t id, uint64_t header,
                                  const void *payload, unsigned length)
{
    
    // Get end point pointer from standard uct_ep_h
    // uct_sci_ep_t is apparently a subclass of uct_ep_h.
    // Basically it specifies uct_ep_h as its parent by having a super element with that function.
    // Then the ucx_derived_of macro is used to get a pointer to uct_sci_ep_t from uct_sci_ep.
    // That macro in turn uses a ucs_container_of macro, which is a typical kind of macro type of macro.
    uct_sci_ep_t* ep       = ucs_derived_of(tl_ep, uct_sci_ep_t);
    sci_packet_t* packet; 
    uct_sci_iface_t* iface = ucs_derived_of(tl_ep->iface, uct_sci_iface_t);
    sci_ctl_t* ctl         = iface->ctls + ep->ctl_offset;
    uint32_t offset = 0; 
    
    if (ep->seq - ctl->ack >= iface->queue_size) {
        return UCS_ERR_NO_RESOURCE;
    }
    
    // Send size is the buffer size (I guess it should have been called packet_size, not send_size
    // Okay, so this is the offset into a larger buffer. It is a rolling window it seems.
    // So there are multiple copy_out buffers in one larger buffer. And it rolls over after reaching queue_size number of copies.
    offset = ep->send_size * (ep->seq % ep->queue_size);
    packet = ep->buf + offset;
    // Don't know where this is checked
    ctl->status = 1;
    
    packet->am_id = id;
    packet->length = length + sizeof(header);
    // Two flushes probably not necessary, un
    uct_am_short_fill_data(ep->buf + offset + sizeof(sci_packet_t), header, payload, length, UCS_ARCH_MEMCPY_NT_DEST);
    SCIFlush(NULL, SCI_FLAG_FLUSH_CPU_BUFFERS_ONLY);    
    packet->status = 1;
    SCIFlush(NULL, SCI_FLAG_FLUSH_CPU_BUFFERS_ONLY);
    ep->seq++; // increment sequence number, will be used next time this is called actually
    DEBUG_PRINT("EP_SEG %d EP_NOD %d AM_ID %d size %d SEQ:%d\n", ep->remote_segment_id, ep->remote_node_id, id, packet->length, ep->seq);
    return UCS_OK;
}
